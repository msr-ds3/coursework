# rank movies by popularity (number of ratings) and compute the cdf, or fraction of all views covered by the top-k movies https://speakerdeck.com/jhofman/modeling-social-data-lecture-2-introduction-to-counting?slide=30
# hint: use dplyr's rank and arrange functions, and the base R sum and cumsum functions
# store the result in a new data frame so you can use it in creating figure 2 from the paper below
ordered_ratings_by_movie <-
  ratings_by_movie %>%
  arrange(desc(num_ratings))
total_ratings <- sum(ratings_by_movie$num_ratings)
movies_cdf <- 
  mutate(ordered_ratings_by_movie, rank = rank(desc(num_ratings), ties.method = "random")) %>%
  mutate(cdf = cumsum(num_ratings)/total_ratings) %>%
  arrange(rank)
# plot the CDF of movie popularity
ggplot(movies_cdf, aes(x = rank, y = cdf)) +
  geom_line() +
  xlab("Movie Rank") +
  ylab("CDF") +
  scale_y_continuous(label = percent)

4 Anatomy of the long tail

# generate the equivalent of figure 2a of this paper:
# note: don't worry about the "null model" lines
# just do the solid lines and dotted line (optional)
# https://5harad.com/papers/long_tail.pdf

num_users <- nrow(ratings_by_user)

least_popular_cdf <-
  inner_join(ratings, movies_cdf) %>% 
  group_by(user_id) %>%
  summarize(one_hundred_percent = max(rank), ninety_percent = quantile(rank, .9), num_ratings = n()) %>%
  pivot_longer(names_to = "percent_satisfied", values_to = "highest_rank", 2:3) %>%
  group_by(percent_satisfied) %>%
  mutate(rank = rank(highest_rank, ties.method = "random"))
## Joining with `by = join_by(movie_id)`
ggplot(least_popular_cdf, aes(x=highest_rank, y=(rank/num_users), color = percent_satisfied)) +
  geom_line() +
  xlab("Inventory Size") +
  ylab("Percent of Users Satisfied")
